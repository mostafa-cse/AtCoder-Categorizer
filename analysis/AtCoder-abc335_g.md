## 1. Problem Statement & Key Idea
Given N integers A_i and a prime P <= 10^{13}, count pairs (i, j) such that A_i^k = A_j mod P for some positive integer k.

## 2. Core Observation & Invariant
In the cyclic group (Z/PZ)*, A_j is a power of A_i iff the cyclic subgroup generated by A_j is contained in that of A_i, which holds iff ord(A_j) divides ord(A_i).

## 3. Step-by-Step Algorithm
1. Factor P - 1. Compute the multiplicative order ord(A_i) for each element in O(log^2 P) with __int128 modular exponentiation.
2. Collect frequencies of all divisors of P - 1.
3. Apply Fast Zeta Transform over the divisor poset to compute F[d] = sum_{c | d} cnt[c]. Sum F[ord(A_i)] for all i.

## 4. Complexity Analysis
- **Time Complexity**: `O(N log^2 P + D log D)`
- **Space Complexity**: `O(D)`

## 5. C++ Source Code
```cpp
#include <iostream>
#include <vector>
#include <map>
#include <algorithm>
#include <cmath>

using namespace std;

long long mul_mod(long long a, long long b, long long m) {
    return (long long)((__int128)a * b % m);
}

long long power_mod(long long base, long long exp, long long mod) {
    long long res = 1;
    base %= mod;
    while (exp > 0) {
        if (exp % 2 == 1) res = mul_mod(res, base, mod);
        base = mul_mod(base, base, mod);
        exp /= 2;
    }
    return res;
}

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);

    int N;
    long long P;
    if (!(cin >> N >> P)) return 0;

    vector<long long> A(N);
    for (int i = 0; i < N; ++i) {
        cin >> A[i];
    }

    long long m = P - 1;
    vector<long long> factors;
    long long temp = m;
    for (long long d = 2; d * d <= temp; ++d) {
        if (temp % d == 0) {
            factors.push_back(d);
            while (temp % d == 0) temp /= d;
        }
    }
    if (temp > 1) factors.push_back(temp);

    auto get_order = [&](long long x) {
        long long o = P - 1;
        for (long long p : factors) {
            while (o % p == 0) {
                if (power_mod(x, o / p, P) == 1) {
                    o /= p;
                } else {
                    break;
                }
            }
        }
        return o;
    };

    vector<long long> orders(N);
    map<long long, int> cnt;
    for (int i = 0; i < N; ++i) {
        orders[i] = get_order(A[i]);
        cnt[orders[i]]++;
    }

    vector<long long> divs;
    for (long long d = 1; d * d <= m; ++d) {
        if (m % d == 0) {
            divs.push_back(d);
            if (d * d != m) divs.push_back(m / d);
        }
    }
    sort(divs.begin(), divs.end());

    int D = (int)divs.size();
    map<long long, int> div_idx;
    for (int i = 0; i < D; ++i) {
        div_idx[divs[i]] = i;
    }

    vector<long long> F(D, 0);
    for (int i = 0; i < D; ++i) {
        auto it = cnt.find(divs[i]);
        if (it != cnt.end()) F[i] = it->second;
    }

    for (long long p : factors) {
        for (int i = 0; i < D; ++i) {
            long long d = divs[i];
            if (d % p == 0) {
                int prev_idx = div_idx[d / p];
                F[i] += F[prev_idx];
            }
        }
    }

    long long ans = 0;
    for (int i = 0; i < N; ++i) {
        int idx = div_idx[orders[i]];
        ans += F[idx];
    }

    cout << ans << "\n";
    return 0;
}
```
