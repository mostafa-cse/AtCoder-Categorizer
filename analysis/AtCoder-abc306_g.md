## 1. Problem Statement & Key Idea
Determine if there is a closed walk from 1 to 1 of length $10^{10^{100}}$ in a directed graph. A vertex $v$ can participate in the walk if and only if it is in the Strongly Connected Component of 1 (reachable from 1 and can reach 1), found via two BFS runs on $G$ and $G^{rev}$. In this induced subgraph, run BFS from 1 to assign depth $d[v]$. The set of cycle lengths is generated by edges $u \to v$ with length $d[u] + 1 - d[v]$.

## 2. Core Observation & Invariant
Any closed walk length is a multiple of $g = \gcd_{(u, v)} |d[u] + 1 - d[v]|$. A walk of length $10^{10^{100}}$ exists if and only if $g > 0$ and $g$ divides $10^{10^{100}}$, which means the only prime factors of $g$ are 2 and 5! If $g$ reduced by factors of 2 and 5 equals 1, answer is Yes, else No.

## 3. Step-by-Step Algorithm
1. 1. Run BFS from vertex 1 on $G$ to find vertices reachable from 1.
2. 2. Run BFS from vertex 1 on $G^{rev}$ to find vertices that can reach 1.
3. 3. Form induced subgraph of vertices in the intersection (SCC of 1).
4. 4. Run BFS from 1 on the induced subgraph to assign tree depth $d[v]$.
5. 5. For each edge $u \to v$ in the induced subgraph, compute $g = \gcd(g, |d[u] + 1 - d[v]|)$.
6. 6. If $g == 0$, output `No`. Else divide out all factors of 2 and 5. Output `Yes` if $g == 1$ else `No`.

## 4. Complexity Analysis
- **Time Complexity**: `O(N + M)`
- **Space Complexity**: `O(N + M)`

## 5. C++ Source Code
```cpp

#include <iostream>
#include <vector>
#include <queue>
#include <numeric>

using namespace std;

long long gcd_ll(long long a, long long b) {
    while (b) {
        a %= b;
        swap(a, b);
    }
    return a;
}

void solve() {
    int n, m;
    if (!(cin >> n >> m)) return;

    vector<vector<int>> adj(n + 1), radj(n + 1);
    for (int i = 0; i < m; ++i) {
        int u, v;
        cin >> u >> v;
        adj[u].push_back(v);
        radj[v].push_back(u);
    }

    // Reachable from 1
    vector<bool> reach1(n + 1, false);
    queue<int> q;
    q.push(1);
    reach1[1] = true;
    while (!q.empty()) {
        int u = q.front();
        q.pop();
        for (int v : adj[u]) {
            if (!reach1[v]) {
                reach1[v] = true;
                q.push(v);
            }
        }
    }

    // Can reach 1
    vector<bool> can_reach1(n + 1, false);
    q.push(1);
    can_reach1[1] = true;
    while (!q.empty()) {
        int u = q.front();
        q.pop();
        for (int v : radj[u]) {
            if (!can_reach1[v]) {
                can_reach1[v] = true;
                q.push(v);
            }
        }
    }

    vector<bool> in_scc(n + 1, false);
    for (int i = 1; i <= n; ++i) {
        in_scc[i] = (reach1[i] && can_reach1[i]);
    }

    // BFS depth in SCC
    vector<int> depth(n + 1, -1);
    depth[1] = 0;
    q.push(1);
    while (!q.empty()) {
        int u = q.front();
        q.pop();
        for (int v : adj[u]) {
            if (in_scc[v] && depth[v] == -1) {
                depth[v] = depth[u] + 1;
                q.push(v);
            }
        }
    }

    long long g = 0;
    for (int u = 1; u <= n; ++u) {
        if (!in_scc[u]) continue;
        for (int v : adj[u]) {
            if (in_scc[v]) {
                long long diff = abs((long long)depth[u] + 1 - depth[v]);
                g = gcd_ll(g, diff);
            }
        }
    }

    if (g == 0) {
        cout << "No\n";
    } else {
        while (g % 2 == 0) g /= 2;
        while (g % 5 == 0) g /= 5;
        if (g == 1) {
            cout << "Yes\n";
        } else {
            cout << "No\n";
        }
    }
}

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);

    int T;
    if (cin >> T) {
        while (T--) solve();
    }
    return 0;
}
```
